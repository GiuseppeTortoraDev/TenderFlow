package com.tenderflow.controller;

import com.tenderflow.entity.User;
import io.quarkus.elytron.security.common.BcryptUtil;
import io.smallrye.jwt.build.Jwt;
import jakarta.annotation.security.PermitAll;
import jakarta.transaction.Transactional;
import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import java.time.Duration;
import java.util.Map;
import java.util.Set;

@Path("/api/auth")
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class AuthResource {

    public static class RegisterRequest {
        public String fullName;
        public String email;
        public String password;
        public String role; // "BUYER" o "SUPPLIER"
    }

    public static class LoginRequest {
        public String email;
        public String password;
    }

    @POST
    @Path("/register")
    @PermitAll
    @Transactional
    public Response register(RegisterRequest req) {
        if (req.email == null || req.password == null || req.role == null) {
            return Response.status(Response.Status.BAD_REQUEST)
                    .entity(Map.of("error", "Compila tutti i campi obbligatori")).build();
        }

        if (User.findByEmail(req.email) != null) {
            return Response.status(Response.Status.CONFLICT)
                    .entity(Map.of("error", "Email già registrata")).build();
        }

        String normalizedRole = req.role.toUpperCase();
        if (!normalizedRole.equals("BUYER") && !normalizedRole.equals("SUPPLIER")) {
            return Response.status(Response.Status.BAD_REQUEST)
                    .entity(Map.of("error", "Ruolo non valido. Usa BUYER o SUPPLIER")).build();
        }

        User user = new User();
        user.fullName = req.fullName != null ? req.fullName : req.email;
        user.email = req.email.toLowerCase().trim();
        user.password = BcryptUtil.bcryptHash(req.password);
        user.role = normalizedRole;
        user.persist();

        String token = generateToken(user);

        return Response.status(Response.Status.CREATED).entity(Map.of(
                "token", token,
                "email", user.email,
                "fullName", user.fullName,
                "role", user.role
        )).build();
    }

    @POST
    @Path("/login")
    @PermitAll
    public Response login(LoginRequest req) {
        if (req.email == null || req.password == null) {
            return Response.status(Response.Status.BAD_REQUEST)
                    .entity(Map.of("error", "Inserisci email e password")).build();
        }

        User user = User.findByEmail(req.email.toLowerCase().trim());
        if (user == null || !BcryptUtil.matches(req.password, user.password)) {
            return Response.status(Response.Status.UNAUTHORIZED)
                    .entity(Map.of("error", "Credenziali non valide")).build();
        }

        String token = generateToken(user);

        return Response.ok(Map.of(
                "token", token,
                "email", user.email,
                "fullName", user.fullName,
                "role", user.role
        )).build();
    }

    private String generateToken(User user) {
        return Jwt.issuer("https://tenderflow.it/issuer")
                .upn(user.email)
                .groups(Set.of(user.role))
                .claim("fullName", user.fullName)
                .expiresIn(Duration.ofHours(24))
                .sign();
    }
}

@Path("/api/tenders/{tenderId}/bids")
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
@Tag(name = "Bid Management", description = "Sottomissione e consultazione offerte per gara")
public class BidResource {

    @Channel("bid-events-out")
    Emitter<Bid> bidEmitter;

    @POST
    @RolesAllowed("SUPPLIER")
    @Transactional
    @SecurityRequirement(name = "jwtAuth")
    @Operation(summary = "Invia offerta (Solo SUPPLIER)", description = "Richiede il ruolo SUPPLIER per sottomettere un'offerta.")
    public Response createBid(
            @Parameter(description = "ID della gara") @PathParam("tenderId") Long tenderId,
            Bid bid) {

        Tender tender = Tender.findById(tenderId);
        if (tender == null) {
            throw new BusinessException("Gara non trovata", Response.Status.NOT_FOUND);
        }

        if (!"OPEN".equalsIgnoreCase(tender.status)) {
            throw new BusinessException("Impossibile inviare offerte per una gara non aperta");
        }

        if (tender.closingDate != null && LocalDate.now().isAfter(tender.closingDate)) {
            throw new BusinessException("Il termine per la presentazione delle offerte per questa gara e' scaduto");
        }

        if (bid.amount == null || bid.amount.compareTo(BigDecimal.ZERO) <= 0) {
            throw new BusinessException("L'importo dell'offerta deve essere maggiore di zero");
        }

        if (tender.budget != null && bid.amount.compareTo(tender.budget) > 0) {
            throw new BusinessException("L'importo dell'offerta supera il budget massimo della gara");
        }

        bid.tender = tender;
        bid.submissionDate = LocalDateTime.now();
        bid.persist();

        bidEmitter.send(bid);

        return Response.status(Response.Status.CREATED).entity(bid).build();
    }

    @GET
    @RolesAllowed({"BUYER", "SUPPLIER"})
    @SecurityRequirement(name = "jwtAuth")
    @Operation(summary = "Recupera offerte di una gara", description = "Accessibile da utenti autenticati come BUYER o SUPPLIER con paginazione e ordinamento.")
    public List<Bid> getBidsForTender(
            @Parameter(description = "ID della gara") @PathParam("tenderId") Long tenderId,
            @Parameter(description = "Numero di pagina (0-indexed)") @QueryParam("page") @DefaultValue("0") int page,
            @Parameter(description = "Elementi per pagina") @QueryParam("size") @DefaultValue("10") int size,
            @Parameter(description = "Campo di ordinamento (es. amount, submissionDate)") @QueryParam("sortBy") @DefaultValue("amount") String sortBy,
            @Parameter(description = "Direzione dell'ordinamento (asc o desc)") @QueryParam("sortOrder") @DefaultValue("asc") String sortOrder) {

        Sort sort = "desc".equalsIgnoreCase(sortOrder) ? Sort.by(sortBy).descending() : Sort.by(sortBy).ascending();

        return Bid.find("tender.id = ?1", sort, tenderId)
                .page(Page.of(page, size))
                .list();
    }
}

@SecurityScheme(securitySchemeName = "jwtAuth", type = SecuritySchemeType.HTTP, scheme = "bearer", bearerFormat = "JWT")
public class TenderResource {

    @Channel("tender-events-out")

    @GET
    @PermitAll
    @Operation(summary = "Lista gare con filtri, paginazione e ordinamento", description = "Endpoint pubblico per la consultazione delle gare d'appalto.")
    public List<Tender> getTenders(
            @Parameter(description = "Stato della gara (es. OPEN, CLOSED)") @QueryParam("status") String status,
            @Parameter(description = "Budget minimo") @QueryParam("minBudget") BigDecimal minBudget,
            @Parameter(description = "Budget massimo") @QueryParam("maxBudget") BigDecimal maxBudget,
            @Parameter(description = "Parola chiave nel titolo") @QueryParam("title") String title,
            @Parameter(description = "Numero di pagina (0-indexed)") @QueryParam("page") @DefaultValue("0") int page,
            @Parameter(description = "Elementi per pagina") @QueryParam("size") @DefaultValue("10") int size,
            @Parameter(description = "Campo per l'ordinamento (es. id, budget, title, closingDate)") @QueryParam("sortBy") @DefaultValue("id") String sortBy,
            @Parameter(description = "Direzione dell'ordinamento (asc o desc)") @QueryParam("sortOrder") @DefaultValue("asc") String sortOrder) {

        StringBuilder query = new StringBuilder("1=1");
        Parameters params = new Parameters();

        if (status != null && !status.isBlank()) {
            query.append(" and status = :status");
            params.and("status", status);
        }
        if (minBudget != null) {
            query.append(" and budget >= :minBudget");
            params.and("minBudget", minBudget);
        }
        if (maxBudget != null) {
            query.append(" and budget <= :maxBudget");
            params.and("maxBudget", maxBudget);
        }
        if (title != null && !title.isBlank()) {
            query.append(" and lower(title) like :title");
            params.and("title", "%" + title.toLowerCase() + "%");
        }

        Sort sort = "desc".equalsIgnoreCase(sortOrder) ? Sort.by(sortBy).descending() : Sort.by(sortBy).ascending();

        return Tender.find(query.toString(), sort, params)
                .page(Page.of(page, size))
                .list();
    }

                    .entity("{\"message\": \"Gara chiusa. Nessuna offerta pervenuta.\", \"tenderId\": " + id + "}")
                    .build();
        }

        return Response.ok(new TenderAwardResult(tender, winningBid)).build();
    }

package com.tenderflow.exception;

import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.ext.Provider;

import java.time.LocalDateTime;

@Provider
public class GlobalExceptionMapper implements ExceptionMapper<Throwable> {

    @Override
    public Response toResponse(Throwable exception) {
        int statusCode = Response.Status.INTERNAL_SERVER_ERROR.getStatusCode();
        String errorName = Response.Status.INTERNAL_SERVER_ERROR.getReasonPhrase();
        String message = exception.getMessage();

        if (exception instanceof BusinessException be) {
            statusCode = be.getStatus().getStatusCode();
            errorName = be.getStatus().getReasonPhrase();
        } else if (exception instanceof WebApplicationException wae) {
            statusCode = wae.getResponse().getStatus();
            Response.Status statusEnum = Response.Status.fromStatusCode(statusCode);
            errorName = statusEnum != null ? statusEnum.getReasonPhrase() : "HTTP Error";
        }

        ErrorResponse errorPayload = new ErrorResponse(
                statusCode,
                errorName,
                message != null ? message : "Si e' verificato un errore inatteso nel sistema.",
                LocalDateTime.now()
        );

        return Response.status(statusCode)
                .type(MediaType.APPLICATION_JSON)
                .entity(errorPayload)
                .build();
    }
}

import { HttpInterceptorFn } from '@angular/common/http';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = localStorage.getItem('jwt_token');

  if (token && !req.url.includes('/api/auth')) {
    const authReq = req.clone({
      setHeaders: {
        Authorization: `Bearer ${token}`
      }
    });
    return next(authReq);
  }

  return next(req);
};

import { Observable, tap } from 'rxjs';

export interface AuthResponse {
  token: string;
  email: string;
  fullName: string;
  role: 'BUYER' | 'SUPPLIER';
}

@Injectable({
  providedIn: 'root'
})
export class AuthService {
  private http = inject(HttpClient);
  private apiUrl = 'http://localhost:8081/api/auth';

  // Signal reattivo per lo stato dell'utente loggato
  currentUser = signal<AuthResponse | null>(this.loadUserFromStorage());
  isAuthenticated = computed(() => this.currentUser() !== null);
  userRole = computed(() => this.currentUser()?.role ?? null);

  private loadUserFromStorage(): AuthResponse | null {
    const raw = localStorage.getItem('tenderflow_user');
    return raw ? JSON.parse(raw) : null;
  }

  register(data: { fullName: string; email: string; password: string; role: string }): Observable<AuthResponse> {
    return this.http.post<AuthResponse>(`${this.apiUrl}/register`, data).pipe(
      tap((res) => this.saveSession(res))
    );
  }

  login(data: { email: string; password: string }): Observable<AuthResponse> {
    return this.http.post<AuthResponse>(`${this.apiUrl}/login`, data).pipe(
      tap((res) => this.saveSession(res))
    );
  }

  private saveSession(res: AuthResponse): void {
    localStorage.setItem('jwt_token', res.token);
    localStorage.setItem('tenderflow_user', JSON.stringify(res));
    this.currentUser.set(res);
  }

  logout(): void {
    localStorage.removeItem('jwt_token');
    localStorage.removeItem('tenderflow_user');
    this.currentUser.set(null);
  }
}

import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { Tender } from '../models/tender.model';

@Injectable({
  providedIn: 'root'
})
export class TenderService {
  private http = inject(HttpClient);
  private readonly apiUrl = 'http://localhost:8081/api/tenders';

  getTenders(): Observable<Tender[]> {
    return this.http.get<Tender[]>(this.apiUrl);
  }

  getTenderById(id: number): Observable<Tender> {
    return this.http.get<Tender>(`${this.apiUrl}/${id}`);
  }

  createTender(tender: Tender): Observable<Tender> {
    return this.http.post<Tender>(this.apiUrl, tender);
  }

  updateStatus(id: number, status: string): Observable<Tender> {
    return this.http.patch<Tender>(`${this.apiUrl}/${id}/status`, { status });
  }
}
